!
hostname P
!

no ip domain lookup
ip domain name lab.local
!
enable secret cisco
username cisco privilege 15 secret cisco
!
service timestamps debug datetime msec
service timestamps log datetime msec
!
ip cef
!

interface Loopback0
 description Router-ID / iBGP source
 ip address 5.5.5.5 255.255.255.255
!
interface GigabitEthernet0/0
 description OSPF to PE1
 ip address 10.0.35.2 255.255.255.252
 ip ospf network point-to-point
 ip ospf hello-interval 3
 ip ospf dead-interval 12
 duplex auto
 speed auto
 media-type rj45
 no shutdown
!

interface GigabitEthernet0/1
 description OSPF to PE2
 ip address 10.0.45.2 255.255.255.252
 ip ospf network point-to-point
 ip ospf hello-interval 3
 ip ospf dead-interval 12
 duplex auto
 speed auto
 media-type rj45
 no shutdown
!

router ospf 1
 router-id 5.5.5.5
 passive-interface default
 no passive-interface GigabitEthernet0/0
 no passive-interface GigabitEthernet0/1
 network 5.5.5.5 0.0.0.0 area 0
 network 10.0.35.0 0.0.0.3 area 0
 network 10.0.45.0 0.0.0.3 area 0
!
router bgp 65000
 bgp router-id 5.5.5.5
 bgp log-neighbor-changes
 neighbor 3.3.3.3 remote-as 65000
 neighbor 3.3.3.3 description PE1-iBGP
 neighbor 3.3.3.3 update-source Loopback0
 neighbor 3.3.3.3 next-hop-self
 neighbor 4.4.4.4 remote-as 65000
 neighbor 4.4.4.4 description PE2-iBGP
 neighbor 4.4.4.4 update-source Loopback0
 neighbor 4.4.4.4 next-hop-self
!

!
line con 0
 exec-timeout 0 0
 logging synchronous
line vty 0 4
 login local
 transport input telnet ssh
!
end
